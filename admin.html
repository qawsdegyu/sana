⚙️ إعدادات سوبابيز (Supabase) والنسخ الاحتياطي

ربط قاعدة بيانات سوبابيز، نسخ كود الـ SQL، استيراد وتصدير نسخة احتياطية من الموقع.

مفاتيح الربط مع مشروع Supabase إعدادات الاتصال

تأكد من أن هذه المفاتيح تطابق مشروعك الحالي في سوبابيز (من Project Settings -> API):

حالة الاتصال وقاعدة البيانات فحص الاتصال...

لتمكين حفظ ومزامنة كافة نصوص وصور الموقع سحابياً في حسابك على Supabase، انسخ كود الـ SQL التالي وشغّله في SQL Editor داخل لوحة تحكم سوبابيز: 

-- 1. إنشاء جدول محتوى الموقع إذا لم يكن موجوداً
CREATE TABLE IF NOT EXISTS public.site_content (
    id TEXT PRIMARY KEY,
    content JSONB NOT NULL,
    updated_at TIMESTAMP WITH TIME ZONE DEFAULT timezone('utc'::text, now()) NOT NULL
);

-- 2. إعطاء الصلاحيات الكاملة لجميع الأدوار
GRANT USAGE ON SCHEMA public TO anon, authenticated, service_role;
GRANT ALL ON TABLE public.site_content TO anon, authenticated, service_role;

-- 3. تعطيل RLS لتسهيل حفظ وتعديل المحتوى فوراً وبدون تعقيد سياسات
ALTER TABLE public.site_content DISABLE ROW LEVEL SECURITY;

-- 4. إدخال أو تحديث سجل الموقع الأولي
INSERT INTO public.site_content (id, content, updated_at)
VALUES ('main_site', '{"status":"connected"}'::jsonb, now())
ON CONFLICT (id) DO UPDATE SET updated_at = now();

-- 5. تأكيد حساب الأدمن فوراً لتجاوز خطأ التفعيل
UPDATE auth.users SET email_confirmed_at = now() WHERE email = 'admin@sana.com';

-- 6. إشعار محرك PostgREST بتحديث كاش الجداول فوراً
NOTIFY pgrst, 'reload schema';
